local nomeArquivo = "ExuHub_ProV2_" .. player.UserId .. ".json"

local function salvarDados()
    local dados = { base = nil, ilhas = {} }
    if posicaoBase then dados.base = {posicaoBase:GetComponents()} end
    for i, ilha in ipairs(ilhas) do
        if ilha.pos then dados.ilhas[i] = {ilha.pos:GetComponents()} end
    end
    if writefile then
        pcall(function() writefile(nomeArquivo, HttpService:JSONEncode(dados)) end)
    end
end

local function carregarDados()
    if readfile and pcall(function() readfile(nomeArquivo) end) then
        local sucesso, conteudo = pcall(function() return readfile(nomeArquivo) end)
        if sucesso then
            local sucessoDec, dados = pcall(function() return HttpService:JSONDecode(conteudo) end)
            if sucessoDec then
                if dados.base then posicaoBase = CFrame.new(unpack(dados.base)) end
                if dados.ilhas then
                    for i, posTbl in pairs(dados.ilhas) do
                        local idx = tonumber(i)
                        if ilhas[idx] then ilhas[idx].pos = CFrame.new(unpack(posTbl)) end
                    end
                end
            end
        end
    end
end

local function criarBotaoCustom(nome, texto, size, posicao, corBase, corBorda)
    local btn = Instance.new("TextButton", container)
    btn.Name = nome
    btn.Size = size
    btn.Position = posicao
    btn.BackgroundColor3 = corBase
    btn.TextColor3 = Color3.fromRGB(245, 245, 255)
    btn.Text = texto
    btn.Font = Enum.Font.GothamMedium
    btn.TextSize = 12
    btn.AutoButtonColor = false

    local corner = Instance.new("UICorner", btn)
    corner.CornerRadius = UDim.new(0, 10)

    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = corBorda or Color3.fromRGB(70, 60, 100)
    stroke.Transparency = 0.5
    stroke.Thickness = 1.2

    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = corBase:Lerp(Color3.fromRGB(255, 255, 255), 0.12)}):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = corBase}):Play()
    end)

    return btn
end

            statusLabel.TextColor3 = Color3.fromRGB(110, 255, 160)

-- Anti-Ataque
btnInvencivel.MouseButton1Click:Connect(function()
    invencivelAtivo = not invencivelAtivo
    if invencivelAtivo then
        btnInvencivel.Text = "🛡️ Anti-Ataque: LIGADO"
        statusLabel.Text = "✨ Proteção contra danos ativa"
        statusLabel.TextColor3 = Color3.fromRGB(210, 140, 255)
        
        conexaoInvencivel = RunService.Stepped:Connect(function()
            local char = player.Character
            if char and char:FindFirstChild("Humanoid") then
                local humanoid = char.Humanoid
                if humanoid.Health < humanoid.MaxHealth then
                    humanoid.Health = humanoid.MaxHealth
                end
            end
        end)
    else
        btnInvencivel.Text = "🛡️ Anti-Ataque: DESLIGADO"
        statusLabel.Text = "🔒 Proteção desativada"
        statusLabel.TextColor3 = Color3.fromRGB(180, 180, 205)
        if conexaoInvencivel then conexaoInvencivel:Disconnect(); conexaoInvencivel = nil end
    end
end)
